use std::collections::BTreeMap;

#[derive(Debug, Clone, PartialEq)]
pub enum Node<'a, T: Ord> {
    Leaf(&'a T),
    Branch { left: Box<Node<'a, T>>, right: Box<Node<'a, T>>, weight: u64 },
}

impl<'a, T: Ord + std::fmt::Debug> Node<'a, T> {
    pub fn depth(&self) -> usize {
        match self {
            Node::Leaf(_) => 1,
            Node::Branch { left, right, .. } => 1 + left.depth().max(right.depth()),
        }
    }
}

fn tally<I: IntoIterator<Item = u32>>(xs: I) -> BTreeMap<u32, usize> {
    xs.into_iter().fold(BTreeMap::new(), |mut acc, x| {
        *acc.entry(x % 16).or_insert(0) += 1;
        acc
    })
}

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let counts = tally([3u32, 19, 7, 35, 7, 3]);
    println!("{counts:?}");
    Ok(())
}
